cmake_minimum_required(VERSION 3.20)
project(ScorchedEverittWin11 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(MSVC)
  add_compile_options(/W4 /fp:strict /utf-8)
  set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
else()
  add_compile_options(-Wall -Wextra -ffp-contract=off)
endif()
enable_testing()
add_executable(core_tests tests/core_tests.cpp)
add_test(NAME source_semantics COMMAND core_tests)
add_executable(render_tests tests/render_tests.cpp)
add_test(NAME renderer_intro COMMAND render_tests)
if(WIN32)
 add_executable(ScorchedEveritt WIN32 src/main.cpp)
 target_compile_definitions(ScorchedEveritt PRIVATE UNICODE _UNICODE)
 target_link_libraries(ScorchedEveritt PRIVATE windowscodecs ole32 winmm gdi32 user32)
 add_custom_command(TARGET ScorchedEveritt POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_directory ${CMAKE_SOURCE_DIR}/picts $<TARGET_FILE_DIR:ScorchedEveritt>/picts)
 add_test(NAME native_smoke COMMAND ScorchedEveritt --smoke-test)
 set_tests_properties(native_smoke PROPERTIES TIMEOUT 60)
endif()
